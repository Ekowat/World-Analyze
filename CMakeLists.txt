cmake_minimum_required(VERSION 3.22)
project(WorldAnalysis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
FetchContent_Declare(preloader GIT_REPOSITORY https://github.com/LiteLDev/preloader-android.git GIT_TAG e0a61cbcc46b7fc84f977958f2bbeac47b6fe856)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1)
FetchContent_Declare(entt GIT_REPOSITORY https://github.com/skypjack/entt.git GIT_TAG v3.15.0)
FetchContent_MakeAvailable(preloader glm entt)
file(GLOB_RECURSE MOD_SOURCES CONFIGURE_DEPENDS src/core/*.cpp src/modules/*.cpp src/config/*.cpp src/launcher/*.cpp)
add_library(WorldAnalysis SHARED src/main.cpp ${MOD_SOURCES})
target_include_directories(WorldAnalysis PUBLIC include PRIVATE src third_party)
target_link_libraries(WorldAnalysis PRIVATE preloader nlohmann_json::nlohmann_json glm::glm EnTT::EnTT fmt::fmt android log EGL GLESv3 GLESv2)
target_compile_options(WorldAnalysis PRIVATE -Oz -fno-rtti -fvisibility=hidden -fvisibility-inlines-hidden -ffunction-sections -fdata-sections)
target_link_options(WorldAnalysis PRIVATE -Wl,--strip-all -Wl,--gc-sections -Wl,--icf=all -Wl,-z,max-page-size=16384)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(TARGET WorldAnalysis POST_BUILD COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/package_levipack.py --library $<TARGET_FILE:WorldAnalysis> --font ${CMAKE_SOURCE_DIR}/resources/minecraft.ttf --version-header ${CMAKE_SOURCE_DIR}/include/worldanalysis/Version.hpp --output $<TARGET_FILE_DIR:WorldAnalysis>/WorldAnalysis.levipack)
